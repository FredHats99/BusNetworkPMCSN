cmake_minimum_required(VERSION 3.21)
project(rome_bus_pmcsn VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(pmcsn_core src/common/audit.c)
target_include_directories(pmcsn_core PUBLIC include)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(rome_bus_pmcsn src/main.c src/ingest/download.c src/validation/keys.c)
target_link_libraries(rome_bus_pmcsn PRIVATE pmcsn_core)
if(WIN32)
  target_link_libraries(rome_bus_pmcsn PRIVATE winhttp)
endif()
if(BUILD_TESTING)
  add_executable(test_audit tests/test_audit.c)
  target_link_libraries(test_audit PRIVATE pmcsn_core)
  add_test(NAME audit_offline COMMAND test_audit)
  add_test(NAME cli_help COMMAND rome_bus_pmcsn --help)
  add_test(NAME csv_stops COMMAND rome_bus_pmcsn audit-static ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/stops.csv)
  add_test(NAME csv_overnight COMMAND rome_bus_pmcsn audit-static ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/stop_times.csv)
  add_test(NAME csv_invalid_coordinates COMMAND rome_bus_pmcsn audit-static ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/invalid_stops.csv)
  set_tests_properties(csv_invalid_coordinates PROPERTIES WILL_FAIL TRUE)
  add_test(NAME checksum_fixture COMMAND ${CMAKE_COMMAND} -DFIXTURE=${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/checksum.txt -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_checksum.cmake)
  add_test(NAME gtfs_references COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:rome_bus_pmcsn> -DWORK=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_keys.cmake)
  if(WIN32)
    add_test(NAME download_preserves_existing COMMAND rome_bus_pmcsn download-static ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/checksum.txt)
    set_tests_properties(download_preserves_existing PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
if(MSVC)
  target_compile_options(pmcsn_core PRIVATE /W4 /WX)
else()
  target_compile_options(pmcsn_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
